/* 1. 读取数据 + 生成 pd target */
data raw;
    set "data.sas7bdat";  /* 你的数据路径 */
    
    * 生成PD和目标变量 ;
    pd = initial_pd * 10000;
    target = ab_perf_bad_n12m;
    keep pd target d_snapshot;
run;

/* ---------------------------------------------------------------------
   关键修复：把日期写成【SAS日期常量】 '30Jun2018'd  而不是 '2018-06-30'
   这样 d_snapshot（数值型）才能和 IN 列表匹配
--------------------------------------------------------------------- */
%let train_snap = '30Jun2014'd, '30Jun2015'd, '30Jun2016'd, '30Jun2017'd, '30Jun2018'd,
                  '30Jun2019'd, '30Jun2020'd, '30Jun2021'd, '30Jun2022'd, '30Jun2023'd;

%let test_snap  = '30Jun2018'd, '30Jun2019'd, '30Jun2020'd, '30Jun2021'd, '30Jun2022'd, '30Jun2023'd;

/* 2. 训练集 */
data df_train;
    set raw;
    where d_snapshot in (&train_snap.);  /* 现在类型完全匹配 */
    
    select(d_snapshot);
        when('30Jun2014'd) ds_idx=1;
        when('30Jun2015'd) ds_idx=2;
        when('30Jun2016'd) ds_idx=3;
        when('30Jun2017'd) ds_idx=4;
        when('30Jun2018'd) ds_idx=5;
        when('30Jun2019'd) ds_idx=6;
        when('30Jun2020'd) ds_idx=7;
        when('30Jun2021'd) ds_idx=8;
        when('30Jun2022'd) ds_idx=9;
        when('30Jun2023'd) ds_idx=10;
        otherwise;
    end;
    
    is_train = 1;
    dataset_id = cats("Tr", ds_idx);
run;

/* 3. 测试集 */
data df_test;
    set raw;
    where d_snapshot in (&test_snap.);  /* 修复完成 */
    
    select(d_snapshot);
        when('30Jun2018'd) ds_idx=11;
        when('30Jun2019'd) ds_idx=12;
        when('30Jun2020'd) ds_idx=13;
        when('30Jun2021'd) ds_idx=14;
        when('30Jun2022'd) ds_idx=15;
        when('30Jun2023'd) ds_idx=16;
        otherwise;
    end;
    
    is_train = 0;
    dataset_id = cats("Ts", ds_idx-10);
run;

/* 4. 合并最终数据 */
data all_data;
    set df_train df_test;
run;
